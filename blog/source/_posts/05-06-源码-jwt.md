package auth

import (
	"crypto/rand"
	"encoding/hex"
	"errors"
	"fmt"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

const (
	TypeAccess  = "access"
	TypeRefresh = "refresh"
)

// Claims 必须嵌入 jwt.RegisteredClaims 才实现 jwt.Claims 接口
type Claims struct {
	UserID   uint64 `json:"user_id"`
	Username string `json:"username"`
	Type     string `json:"type"`
	jwt.RegisteredClaims
}

// refresh标识
// 面试点
// jti 是什么？ JWT ID，给每个 token 一个唯一编号。

// 为什么 refresh 需要它？
// JWT 签出去就无法收回（无状态）
// 要支持「登出」「token 被盗后撤销」，必须有服务端记录
// 所以：refresh 的 jti 存进 Redis（白名单），校验时查 Redis；删掉 jti = 该 token 立即失效
// access token 不用 jti，因为它短命（2h），且频繁签发

// 为什么用 crypto/rand 而不是 math/rand？
// crypto/rand 是密码学安全随机数，不可预测
// math/rand 是伪随机，可被预测 → jti 可被猜出 → 安全漏洞
// 这是安全相关代码的铁律
func GenerateJit() (string, error) {
	b := make([]byte, 16)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return hex.EncodeToString(b), nil
}

func GenerateAccess(secret string, userID uint64, username string, expireMinutes int) (string, error) {
	now := time.Now()

	claims := Claims{
		UserID:   userID,
		Username: username,
		Type:     TypeAccess,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(now.Add(time.Duration(expireMinutes) * time.Minute)),
			IssuedAt:  jwt.NewNumericDate(now),
		},
	}
	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
	return token.SignedString([]byte(secret))
}

// ① 为什么 refresh 长命（30d）？
// 它只在刷新时用，不常暴露
// 长命才能"30 天内用过就不掉登录"
// 泄露风险靠 jti 白名单控制（可撤销）
// ② 为什么返回 jti？
// 调用方（登录 service）要把 refresh:{userID}:{jti} 写进 Redis
// 这样服务端有了"这个 refresh 有效"的记录
// ③ 为什么 jti 放 RegisteredClaims.ID？

// JWT 标准里 ID 字段就对应 jti，不用自己仿造
// 解析时 claims.ID 直接能拿到
func GenerateRefresh(secret string, userID uint64, expireHours int) (string, string, error) {
	jti, err := GenerateJit()
	if err != nil {
		return "", "", err
	}
	now := time.Now()
	claims := Claims{
		UserID: userID,
		Type:   TypeRefresh,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(now.Add(time.Duration(expireHours) * time.Hour)),
			IssuedAt:  jwt.NewNumericDate(now),
			ID:        jti, // 把 jti 放进标准字段 ID
		},
	}
	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
	s, err := token.SignedString([]byte(secret))
	if err != nil {
		return "", "", err
	}
	return s, jti, nil
}

func Parse(secret, tokenStr string) (*Claims, error) {
	token, err := jwt.ParseWithClaims(tokenStr, &Claims{}, func(t *jwt.Token) (any, error) {
		// 关键安全校验：算法必须是 HMAC
		if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
			return nil, fmt.Errorf("unexpected signing method: %v", t.Header["alg"])
		}
		return []byte(secret), nil
	})
	if err != nil {
		return nil, err
	}
	claims, ok := token.Claims.(*Claims)
	if !ok || !token.Valid {
		return nil, errors.New("invalid token")
	}
	return claims, nil
}

// ① ParseWithClaims(tokenStr, &Claims{}, keyFunc) 的第二个参数为什么传指针？

// 解析时要把 payload 反序列化写进这个变量 → 必须传 &Claims{}
// 传值的话解析结果丢了
// ② keyFunc 是什么？

// 一个回调：GORM... 不，jwt 库解析时需要你告诉它"用哪个密钥验签"
// 返回 []byte(secret) 即可
// ③ 为什么 keyFunc 里要校验算法？—— 这是必考点

// 不校验的话有 算法混淆攻击：
// 攻击者把 header 的 alg 改成 none，signature 留空
// 或把 HS256 改成 RS256，用你的公钥当 HMAC 密钥
// 校验 t.Method 是 *jwt.SigningMethodHMAC 就堵住了这两条路
// 面试问"JWT 有什么安全风险"必答这一条
// ④ 为什么还要检查 token.Valid？

// ParseWithClaims 返回 err == nil 时通常 valid，但显式检查更稳
// 也顺便防了 claims 类型断言失败的情况
// ⑤ 注意：Parse 不检查 Type

// 这是故意的：把它留给调用方
// 中间件检查 type == "access"，refresh 接口检查 type == "refresh"
// 为什么要分开？因为同一个 Parse 要服务两种 token，把"用途校验"交给最清楚用途的那一层
